    '/news/bels-begin-surrey-league-title-defence-with-victory':[IMG+'sl-2026-m1-men.jpg','A Belgrave man leading a group of runners at the first Surrey League match in Wimbledon. Photo: George Parker','50% 50%'],
    '/news/a-very-good-day-for-the-club':[IMG+'sl-2026-m1-women.jpg','A Belgrave woman, number 45, racing at the first Surrey League match in Wimbledon. Photo: George Parker','50% 50%']
  };
  var MON=['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  function esc(s){return String(s==null?'':s).replace(/&amp;/g,'&').replace(/[&<>"]/g,function(c){return{'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]})}

  function parts(s){var p=s.split('-');return{y:+p[0],m:+p[1]-1,d:+p[2]}}
  function csvLine(l){var v=[],c='',q=false;for(var i=0;i<l.length;i++){var ch=l[i];if(ch==='"')q=!q;else if(ch===','&&!q){v.push(c.trim());c=''}else c+=ch}v.push(c.trim());return v}
  function parseSheet(t){var L=t.trim().split(/\r?\n/),h=csvLine(L[0]).map(function(x){return x.toLowerCase()});return L.slice(1).map(function(l){var v=csvLine(l),o={};h.forEach(function(k,i){o[k]=v[i]||''});return o}).filter(function(o){return /^\d{4}-\d\d-\d\d$/.test(o.date)&&o.name})}
  function timeout(p,ms){return Promise.race([p,new Promise(function(_,r){setTimeout(function(){r(new Error('timeout'))},ms)})])}
  function img(it,w){var a=it.assetUrl||'';if(!a)return '';return a+(a.indexOf('?')<0?'?format='+w+'w':'')}

  function hasTag(it,t){return (it.tags||[]).some(function(x){return x.toLowerCase()===t})}
  function team(it){var c=(it.categories||[]).join('|').toLowerCase();if(c.indexOf('junior')>=0)return 'Juniors';if(c.indexOf('women')>=0)return 'Women';if(c.indexOf('mens')>=0)return 'Men';return ''}
  var ICON={done:'<svg width="18" height="18" viewBox="0 0 18 18" role="img" aria-label="Done"><circle cx="9" cy="9" r="8" fill="#d6b97a"/><path d="M5 9.2l2.6 2.6L13 6.4" stroke="#7a001d" stroke-width="2" fill="none"/></svg>',
    next:'<svg width="18" height="18" viewBox="0 0 18 18" role="img" aria-label="Next"><circle cx="9" cy="9" r="7.5" fill="none" stroke="#d6b97a" stroke-width="2"/><circle cx="9" cy="9" r="3.5" fill="#d6b97a"/></svg>',
    later:'<svg width="18" height="18" viewBox="0 0 18 18" aria-hidden="true"><circle cx="9" cy="9" r="7.5" fill="none" stroke="rgba(255,255,255,.55)" stroke-width="1.5"/></svg>'};

  function render(news,sheet){
    var today=iso(new Date());
    var matches=sheet.filter(function(r){return /^surrey league match \d/i.test(r.name)&&r.date>=SEASON_FROM&&r.date<=SEASON_TO})

    var next=null;matches.forEach(function(m){m.done=m.date<today;if(!m.done&&!next)next=m});
    var trackerHTML=matches.map(function(m){var p=parts(m.date),st=m.done?'done':(m===next?'next':'later');
      var what=m.done?'<span>'+(RESULTS[m.n]||'Result to follow').split(' · ').map(esc).join('<span class="bh-sep"> · </span>')+'</span>':esc(m.venue)+(m===next?'<span class="bh-pill">Next</span>':'');
      return '<li class="bh-match is-'+st+'"><span class="bh-ic">'+ICON[st]+'</span><span class="bh-when"><b>Match '+m.n+'</b><span>'+p.d+' '+MON[p.m]+'<span class="bh-yr"> '+p.y+'</span></span></span><span class="bh-what">'+what+'</span></li>'}).join('');
    var from=new Date(SEASON_FROM).getTime(),to=new Date(SEASON_TO+'T23:59:59').getTime();
    var order={Men:0,Women:1,Juniors:2,'':3};
    var def=news.filter(function(it){return hasTag(it,'surrey league')&&it.publishOn>=from&&it.publishOn<=to}).map(function(it){
      var d=iso(new Date(it.publishOn)),m=null;matches.forEach(function(x){if(x.date<=d)m=x});
      return{it:it,m:m,team:team(it)}}).sort(function(a,b){var an=a.m?a.m.n:0,bn=b.m?b.m.n:0;if(an!==bn)return bn-an;if(order[a.team]!==order[b.team])return order[a.team]-order[b.team];return b.it.publishOn-a.it.publishOn});
    var used={};
    var cards=def.map(function(x){var it=x.it;used[it.id]=1;var ph=PHOTOS[it.fullUrl],src=ph?ph[0]:img(it,750),alt=ph?ph[1]:'',pos=ph?ph[2]:focal(it);
      var label=[x.m?'Match '+x.m.n:'Surrey League',x.team].filter(Boolean).join(' · ');
      return '<li><a class="bh-dcard" href="'+esc(it.fullUrl)+'">'+(src?'<img src="'+esc(src)+'" alt="'+esc(alt)+'" loading="lazy" style="object-position:'+pos+'">':'')+'<span class="bh-txt"><span class="bh-eyebrow">'+esc(label)+'</span><span class="bh-t">'+esc(it.title)+'</span></span></a></li>'}).join('');
    if(next){var np=parts(next.date);cards+='<li class="bh-dnext"><span class="bh-eyebrow">Match '+next.n+' · '+np.d+' '+MON[np.m]+'</span><span class="bh-t">Reports from '+esc(next.venue)+' will appear here</span></li>'}
    var up=sheet.filter(function(r){return r.date>=today}).sort(function(a,b){return a.date<b.date?-1:1}).slice(0,3);
    var fxHTML=up.map(function(r){var p=parts(r.date);return '<li><span class="bh-d"><b>'+p.d+'</b><span>'+MON[p.m]+'<span class="bh-yr"> '+p.y+'</span></span></span><span class="bh-n"><b>'+esc(r.name.replace(/^Surrey League (Match \d)/i,'Surrey League · $1'))+'</b><span>'+esc(r.venue)+'</span></span></li>'}).join('');
    var more=news.filter(function(it){return !used[it.id]}).slice(0,4).map(function(it){var p=new Date(it.publishOn),s=img(it,750);
      return '<li><a class="bh-mcard" href="'+esc(it.fullUrl)+'">'+(s?'<img src="'+esc(s)+'" alt="" loading="lazy" style="object-position:'+focal(it)+'">':'')+'<span class="bh-txt"><span class="bh-t">'+esc(it.title)+'</span><span class="bh-eyebrow">'+p.getDate()+' '+MON[p.getMonth()]+' '+p.getFullYear()+'</span></span></a></li>'}).join('');
    var h='';
    h+='<section class="bh-def" aria-labelledby="bh-def-h"><div class="bh-in"><div class="bh-top"><div class="bh-intro"><p class="bh-eyebrow">Surrey League · Winter 2026/27</p><h2 id="bh-def-h">Defending the double</h2><p class="bh-lede">Our men and women are the reigning Surrey League champions. Four matches decide this winter’s titles.</p></div>';
    if(trackerHTML)h+='<ol class="bh-matches" aria-label="Surrey League matches">'+trackerHTML+'</ol>';
    h+='</div>';

  }

    if(!anchor||document.querySelector('.bh-home'))return;

  }

  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',build);else build();
